# Notebook interface
# Interactive Python shell used inside JupyterLab.
ipython
# Connects the Python kernel to the Jupyter interface so cells can run.
ipykernel
# Main notebook interface — open problems.ipynb with: jupyter lab
jupyterlab
# Classic notebook server; also required by some JupyterLab extensions.
notebook

# Core scientific computing
# Numerical arrays and matrix operations — used in Problems 1–5 for
# probability vectors, gate matrices, and output classification.
numpy
# Scientific computing utilities — used in Problems 1–2 for
# mathematical helper functions (combinatorics, linear algebra).
scipy
# Symbolic mathematics — used in Problems 2–3 to keep expressions like
# 1/√2 exact rather than floating-point approximations.
sympy
# Plotting library — used in Problems 3–5 to draw circuit diagrams
# and measurement histograms.
matplotlib

# Quantum computing
# IBM's quantum framework — used in Problems 3–5 to build and draw
# quantum circuits (QuantumCircuit, gates, transpile).
# The [visualization] extra adds circuit diagram rendering via matplotlib.
qiskit[visualization]

# IBM's local quantum simulator — used in Problems 3–5 to run circuits
# without real quantum hardware (AerSimulator).
qiskit-aer
